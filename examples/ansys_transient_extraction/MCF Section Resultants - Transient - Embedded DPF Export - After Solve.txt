# Embedded MCF section-resultant export.
# Target Callback must be "On After Solve".
# This code intentionally avoids importing the external reconstructor .py file.
# It uses Mechanical-hosted DPF force_summation for modal section force/moment
# coefficients, parses the MSUP file.mcf, and writes global/local CSV output.

def after_solve(this, analysis):# Do not edit this line
    import csv
    import json
    import math
    import os
    import re
    import traceback

    ELEMENT_NS = "DENEME"
    LOCAL_CSYS_NAME = "Coordinate System"
    OUTPUT_BASENAME = "section_resultants_from_tree_embedded_latest"
    FORCE_TYPE_STATIC = 1
    SECTION_NORMAL_AXIS = "z"
    SECTION_EXTRACTION_SIDE = "positive"
    SIDE_FILTER_TOLERANCE = 1.0e-8
    SECTION_SCOPE_METHOD = "construction_surface_cut_elements_side_nodes"
    MOMENT_REFERENCE_MODE = "mechanical_probe_mesh_centroid"
    # User/probe convention: positive side is the selected coordinate system's
    # local +Z side. Match Mechanical's Surface probe extraction by using
    # elements cut by that plane and nodes on the requested side of those
    # elements, instead of relying on a cut-plane node named selection.
    # Mechanical's probe Summation=Centroid is matched by summing moments about
    # the centroid of the mesh side used by the probe. For an offset
    # construction surface, the DPF force_summation side that matches
    # Mechanical's force sign is opposite to the mesh-side centroid used by
    # Mechanical for the moment summation point. If the cut is mesh-aligned,
    # fall back to the on-plane nodes.

    def safe_text(value):
        try:
            if value is None:
                return None
            return str(value)
        except Exception:
            return None

    def scalar(value):
        for candidate in (value, getattr(value, "Value", None)):
            try:
                if candidate is not None:
                    return float(candidate)
            except Exception:
                pass
        match = re.search(r"[-+]?\d+(?:\.\d*)?(?:[eE][-+]?\d+)?", safe_text(value) or "")
        if match:
            return float(match.group(0))
        raise ValueError("Could not convert {0!r} to float".format(value))

    def vector3(value):
        result = []
        for attr_index, attr_name in enumerate(("X", "Y", "Z")):
            try:
                result.append(scalar(value[attr_index]))
                continue
            except Exception:
                pass
            result.append(scalar(getattr(value, attr_name)))
        return result

    def norm3(values):
        length = math.sqrt(sum(float(v) * float(v) for v in values))
        if length <= 0.0:
            raise ValueError("Zero-length coordinate-system axis.")
        return [float(v) / length for v in values]

    def cross(a, b):
        return [
            a[1] * b[2] - a[2] * b[1],
            a[2] * b[0] - a[0] * b[2],
            a[0] * b[1] - a[1] * b[0],
        ]

    def dot(a, b):
        return sum(float(x) * float(y) for x, y in zip(a, b))

    def find_analysis(name):
        for item in list(DataModel.GetObjectsByType(DataModelObjectCategory.Analysis)):
            if (safe_text(getattr(item, "Name", "")) or "").lower() == name.lower():
                return item
        raise ValueError("Analysis not found: {0}".format(name))

    def find_coordinate_system(name):
        for item in list(Model.CoordinateSystems.Children):
            if (safe_text(getattr(item, "Name", "")) or "").lower() == name.lower():
                origin = vector3(getattr(item, "Origin"))
                x_axis = norm3(vector3(getattr(item, "XAxis")))
                y_axis = norm3(vector3(getattr(item, "YAxis")))
                z_axis = norm3(vector3(getattr(item, "ZAxis")))
                determinant = (
                    x_axis[0] * (y_axis[1] * z_axis[2] - y_axis[2] * z_axis[1])
                    - x_axis[1] * (y_axis[0] * z_axis[2] - y_axis[2] * z_axis[0])
                    + x_axis[2] * (y_axis[0] * z_axis[1] - y_axis[1] * z_axis[0])
                )
                if abs(abs(determinant) - 1.0) > 1.0e-5:
                    raise ValueError(
                        "Coordinate system {0!r} axes are not orthonormal enough; determinant={1}".format(
                            name, determinant
                        )
                    )
                return {
                    "name": safe_text(getattr(item, "Name", "")),
                    "object_id": safe_text(getattr(item, "ObjectId", "")),
                    "origin": origin,
                    "axes": [x_axis, y_axis, z_axis],
                    "determinant": determinant,
                    "coordinate_system_type": safe_text(getattr(item, "CoordinateSystemType", "")),
                }
        raise ValueError("Coordinate system not found: {0}".format(name))

    def get_named_scoping(model, requested_name):
        names = [safe_text(name) for name in list(model.AvailableNamedSelections)]
        for name in names:
            if name and name.lower() == requested_name.lower():
                return model.GetNamedSelection(name), name
        raise ValueError(
            "Named selection {0!r} not found in modal RST. Available: {1}".format(
                requested_name, ", ".join([n or "" for n in names])
            )
        )

    def parse_mcf(path):
        expected_modes = None
        times = []
        modal_coordinates = []
        current_time = None
        current_coordinates = []
        in_table = False
        record_start_line = None
        number_pattern = re.compile(
            r"[+-]?(?:\d+(?:\.\d*)?|\.\d+)(?:[ED][+-]?\d+)?",
            re.IGNORECASE,
        )

        def parse_float_tokens(text):
            return [
                float(token.replace("D", "E").replace("d", "E"))
                for token in number_pattern.findall(text)
            ]

        def finish_record(line_number):
            if current_time is None:
                return
            if expected_modes is not None and len(current_coordinates) != expected_modes:
                raise ValueError(
                    "MCF record ending near line {0} has {1} modal coordinates; "
                    "expected {2}. If MCFOPT wrapping is enabled, continuation "
                    "lines must be present.".format(
                        line_number, len(current_coordinates), expected_modes
                    )
                )
            if expected_modes is None:
                inferred = len(current_coordinates)
                if inferred <= 0:
                    raise ValueError(
                        "MCF record ending near line {0} has no modal coordinates.".format(
                            line_number
                        )
                    )
            times.append(current_time)
            modal_coordinates.append(list(current_coordinates))

        with open(path, "r") as stream:
            for line_number, line in enumerate(stream, 1):
                stripped = line.strip()
                lower = stripped.lower()
                if not stripped:
                    continue
                if lower.startswith("number of modes"):
                    values = parse_float_tokens(stripped)
                    if values:
                        expected_modes = int(values[-1])
                    continue
                if "time" in lower and "coordinate" in lower:
                    in_table = True
                    continue
                if not in_table:
                    continue
                values = parse_float_tokens(stripped)
                if not values:
                    continue
                starts_record = bool(re.match(r"^[+-]?(?:\d+(?:\.\d*)?|\.\d+)[ED][+-]?\d+", stripped, re.IGNORECASE))
                if starts_record:
                    finish_record(line_number - 1)
                    current_time = values[0]
                    current_coordinates = list(values[1:])
                    record_start_line = line_number
                elif current_time is not None:
                    current_coordinates.extend(values)
                if (
                    expected_modes is not None
                    and current_time is not None
                    and len(current_coordinates) > expected_modes
                ):
                    raise ValueError(
                        "MCF record near line {0} contains {1} modal coordinates; "
                        "expected {2}. The modal-coordinate wrapping could not be "
                        "parsed unambiguously.".format(
                            record_start_line, len(current_coordinates), expected_modes
                        )
                    )
        finish_record(line_number if "line_number" in locals() else 0)
        if not times:
            raise ValueError("No modal-coordinate rows were found in {0}".format(path))
        row_lengths = set(len(row) for row in modal_coordinates)
        if len(row_lengths) != 1:
            raise ValueError(
                "MCF modal-coordinate rows have inconsistent lengths: {0}. The file "
                "may use wrapped MCFOPT output but does not include a readable "
                "'Number of Modes' header.".format(sorted(row_lengths))
            )
        return times, modal_coordinates

    def dpf_node_xyz(node):
        return [float(node.X), float(node.Y), float(node.Z)]

    def centroid(points):
        if not points:
            raise ValueError("Cannot calculate centroid of an empty point set.")
        return [
            sum(float(point[index]) for point in points) / float(len(points))
            for index in range(3)
        ]

    def dpf_element_node_ids(element):
        return [int(node_id) for node_id in list(element.NodeIds)]

    def element_centroid(mesh, element_id):
        element = mesh.ElementById(int(element_id))
        node_ids = dpf_element_node_ids(element)
        if not node_ids:
            raise ValueError("Element {0} has no nodes.".format(element_id))
        coordinates = [dpf_node_xyz(mesh.NodeById(node_id)) for node_id in node_ids]
        return [
            sum(point[index] for point in coordinates) / float(len(coordinates))
            for index in range(3)
        ], node_ids

    def axis_from_coordinate_system(local_csys, axis_name):
        axis_name = (axis_name or "").lower()
        axis_index = {"x": 0, "y": 1, "z": 2}.get(axis_name)
        if axis_index is None:
            raise ValueError("Unsupported section normal axis: {0!r}".format(axis_name))
        return local_csys["axes"][axis_index]

    def node_signed_distance(mesh, node_id, origin, normal):
        xyz = dpf_node_xyz(mesh.NodeById(int(node_id)))
        return dot(
            [
                xyz[0] - origin[0],
                xyz[1] - origin[1],
                xyz[2] - origin[2],
            ],
            normal,
        )

    def construction_surface_scoping(dpf_module, model, elem_scope, local_csys):
        mesh = model.Mesh
        normal = axis_from_coordinate_system(local_csys, SECTION_NORMAL_AXIS)
        origin = local_csys["origin"]
        raw_ids = [int(element_id) for element_id in list(elem_scope.Ids)]
        cut_element_ids = []
        not_cut_element_ids = []
        selected_node_ids = set()
        positive_node_ids = set()
        negative_node_ids = set()
        on_plane_node_ids = set()
        element_failures = []
        requested_positive = SECTION_EXTRACTION_SIDE.lower().startswith("pos")
        requested_negative = SECTION_EXTRACTION_SIDE.lower().startswith("neg")

        for element_id in raw_ids:
            try:
                element = mesh.ElementById(int(element_id))
                node_ids = dpf_element_node_ids(element)
                signed_distances = [
                    node_signed_distance(mesh, node_id, origin, normal)
                    for node_id in node_ids
                ]
            except Exception as exc:
                element_failures.append({"element_id": element_id, "error": repr(exc)})
                continue

            if not signed_distances:
                element_failures.append({"element_id": element_id, "error": "Element has no nodes."})
                continue

            min_distance = min(signed_distances)
            max_distance = max(signed_distances)
            if min_distance > SIDE_FILTER_TOLERANCE or max_distance < -SIDE_FILTER_TOLERANCE:
                not_cut_element_ids.append(element_id)
                continue

            cut_element_ids.append(element_id)
            for node_id, signed_distance in zip(node_ids, signed_distances):
                if signed_distance > SIDE_FILTER_TOLERANCE:
                    positive_node_ids.add(int(node_id))
                elif signed_distance < -SIDE_FILTER_TOLERANCE:
                    negative_node_ids.add(int(node_id))
                else:
                    on_plane_node_ids.add(int(node_id))

                if requested_positive:
                    if signed_distance >= -SIDE_FILTER_TOLERANCE:
                        selected_node_ids.add(int(node_id))
                elif requested_negative:
                    if signed_distance <= SIDE_FILTER_TOLERANCE:
                        selected_node_ids.add(int(node_id))
                else:
                    selected_node_ids.add(int(node_id))

        if not cut_element_ids:
            raise ValueError(
                "The construction-surface section cut selected zero elements. side={0}, raw_count={1}, not_cut={2}".format(
                    SECTION_EXTRACTION_SIDE,
                    len(raw_ids),
                    len(not_cut_element_ids),
                )
            )
        if not selected_node_ids:
            raise ValueError(
                "The construction-surface side filter selected zero nodes. side={0}, cut_element_count={1}".format(
                    SECTION_EXTRACTION_SIDE,
                    len(cut_element_ids),
                )
            )

        element_scope = dpf_module.Scoping()
        element_scope.Ids = cut_element_ids
        node_scope = dpf_module.Scoping()
        selected_node_ids_sorted = sorted(selected_node_ids)
        node_scope.Ids = selected_node_ids_sorted
        positive_node_ids_sorted = sorted(positive_node_ids)
        negative_node_ids_sorted = sorted(negative_node_ids)
        on_plane_node_ids_sorted = sorted(on_plane_node_ids)
        selected_node_points = [
            dpf_node_xyz(mesh.NodeById(int(node_id))) for node_id in selected_node_ids_sorted
        ]
        selected_node_centroid = centroid(selected_node_points)
        selected_signed_distances = [
            dot(
                [
                    point[0] - origin[0],
                    point[1] - origin[1],
                    point[2] - origin[2],
                ],
                normal,
            )
            for point in selected_node_points
        ]
        if requested_positive:
            reference_node_ids = negative_node_ids_sorted or on_plane_node_ids_sorted or selected_node_ids_sorted
            reference_node_side = "opposite_negative_for_positive_probe"
        elif requested_negative:
            reference_node_ids = positive_node_ids_sorted or on_plane_node_ids_sorted or selected_node_ids_sorted
            reference_node_side = "opposite_positive_for_negative_probe"
        else:
            reference_node_ids = on_plane_node_ids_sorted or selected_node_ids_sorted
            reference_node_side = "on_plane_or_selected_for_all_sides"
        reference_node_points = [
            dpf_node_xyz(mesh.NodeById(int(node_id))) for node_id in reference_node_ids
        ]
        moment_reference_centroid = centroid(reference_node_points)
        return element_scope, node_scope, {
            "enabled": True,
            "method": SECTION_SCOPE_METHOD,
            "normal_axis": SECTION_NORMAL_AXIS,
            "extraction_side": SECTION_EXTRACTION_SIDE,
            "tolerance": SIDE_FILTER_TOLERANCE,
            "raw_element_count": len(raw_ids),
            "cut_element_count": len(cut_element_ids),
            "selected_node_count": len(selected_node_ids),
            "selected_node_centroid": selected_node_centroid,
            "selected_node_signed_distance_min": min(selected_signed_distances),
            "selected_node_signed_distance_max": max(selected_signed_distances),
            "moment_reference_centroid": moment_reference_centroid,
            "moment_reference_node_side": reference_node_side,
            "moment_reference_node_count": len(reference_node_ids),
            "moment_reference_node_ids_preview": reference_node_ids[:30],
            "positive_node_count": len(positive_node_ids),
            "negative_node_count": len(negative_node_ids),
            "on_plane_node_count": len(on_plane_node_ids),
            "not_cut_element_count": len(not_cut_element_ids),
            "element_failure_count": len(element_failures),
            "cut_element_ids_preview": cut_element_ids[:30],
            "selected_node_ids_preview": selected_node_ids_sorted[:30],
            "positive_node_ids_preview": positive_node_ids_sorted[:30],
            "negative_node_ids_preview": negative_node_ids_sorted[:30],
            "on_plane_node_ids_preview": on_plane_node_ids_sorted[:30],
            "not_cut_element_ids_preview": not_cut_element_ids[:30],
            "element_failures_preview": element_failures[:10],
        }

    def force_moment_coefficients_with_dpf(modal_rst, reference_xyz, local_csys):
        import mech_dpf
        import Ans.DataProcessing as dpf

        mech_dpf.setExtAPI(ExtAPI)
        data_sources = dpf.DataSources(modal_rst)
        model = dpf.Model(data_sources)
        elem_scope, elem_name = get_named_scoping(model, ELEMENT_NS)
        filtered_elem_scope, node_scope, side_filter_info = construction_surface_scoping(
            dpf, model, elem_scope, local_csys
        )
        moment_reference_xyz = list(reference_xyz)
        if MOMENT_REFERENCE_MODE == "mechanical_probe_mesh_centroid":
            moment_reference_xyz = [
                float(value) for value in side_filter_info["moment_reference_centroid"]
            ]
        elif MOMENT_REFERENCE_MODE == "selected_side_node_centroid":
            moment_reference_xyz = [
                float(value) for value in side_filter_info["selected_node_centroid"]
            ]
        elif MOMENT_REFERENCE_MODE != "coordinate_system_origin":
            raise ValueError(
                "Unsupported MOMENT_REFERENCE_MODE: {0!r}".format(MOMENT_REFERENCE_MODE)
            )

        number_sets = int(model.TimeFreqSupport.NumberSets)
        if number_sets <= 0:
            raise ValueError("Modal RST has no time/frequency sets: {0}".format(modal_rst))
        time_scoping = dpf.Scoping()
        time_scoping.Ids = list(range(1, number_sets + 1))

        op = dpf.operators.averaging.force_summation()
        op.inputs.data_sources.Connect(data_sources)
        op.inputs.time_scoping.Connect(time_scoping)
        op.inputs.nodal_scoping.Connect(node_scope)
        op.inputs.elemental_scoping.Connect(filtered_elem_scope)
        op.inputs.force_type.Connect(FORCE_TYPE_STATIC)

        force_fc = op.outputs.force_accumulation.GetData()
        moment_fc = op.outputs.moment_accumulation.GetData()
        ref_cross_adjustment = moment_reference_xyz
        coefficients = []
        for mode_id in range(1, number_sets + 1):
            force = [float(value) for value in list(force_fc.GetFieldByTimeId(mode_id).Data)[:3]]
            moment_about_origin = [
                float(value) for value in list(moment_fc.GetFieldByTimeId(mode_id).Data)[:3]
            ]
            ref_cross_force = cross(ref_cross_adjustment, force)
            moment_about_reference = [
                moment_about_origin[index] - ref_cross_force[index] for index in range(3)
            ]
            coefficients.append(force + moment_about_reference)

        return {
            "coefficients": coefficients,
            "mode_count": number_sets,
            "element_named_selection": elem_name,
            "surface_node_named_selection": None,
            "surface_node_scope_method": SECTION_SCOPE_METHOD,
            "element_count": int(len(filtered_elem_scope.Ids)),
            "raw_element_count": int(len(elem_scope.Ids)),
            "surface_node_count": int(len(node_scope.Ids)),
            "side_filter": side_filter_info,
            "requested_reference_xyz": reference_xyz,
            "moment_reference_mode": MOMENT_REFERENCE_MODE,
            "moment_reference_xyz": moment_reference_xyz,
            "modal_available_named_selections": names_as_list(model.AvailableNamedSelections),
        }

    def names_as_list(names):
        return [safe_text(name) for name in list(names)]

    def rotate_to_local(global_row, axes):
        force = global_row[:3]
        moment = global_row[3:]
        return [
            dot(force, axes[0]),
            dot(force, axes[1]),
            dot(force, axes[2]),
            dot(moment, axes[0]),
            dot(moment, axes[1]),
            dot(moment, axes[2]),
        ]

    def max_abs_by_component(rows):
        labels = ["fx", "fy", "fz", "mx", "my", "mz"]
        result = {}
        for index, label in enumerate(labels):
            result[label] = max([abs(row[index]) for row in rows] or [0.0])
        return result

    def json_default(value):
        return safe_text(value)

    def json_safe(value):
        try:
            string_types = (basestring,)
        except NameError:
            string_types = (str,)
        try:
            integer_types = (int, long)
        except NameError:
            integer_types = (int,)
        if value is None or isinstance(value, (bool, float) + integer_types + string_types):
            return value
        if isinstance(value, (list, tuple)):
            return [json_safe(item) for item in value]
        if isinstance(value, dict):
            result = {}
            for key, item in value.items():
                result[safe_text(key)] = json_safe(item)
            return result
        return json_default(value)

    def write_summary(path, payload):
        with open(path, "w") as stream:
            stream.write(json.dumps(json_safe(payload), indent=2, sort_keys=True))
            stream.write("\n")

    def solver_root_from_working_dir(working_dir):
        root = os.path.abspath(working_dir)
        for _ in range(6):
            root = os.path.dirname(root)
        return root

    def find_current_mcf(project_root, expected_path, modal_rst):
        if os.path.isfile(expected_path):
            return expected_path, "analysis_working_dir"
        candidates = []
        for current_root, _dirs, files in os.walk(project_root):
            for file_name in files:
                if file_name.lower() == "file.mcf":
                    full_path = os.path.join(current_root, file_name)
                    try:
                        candidates.append((os.path.getmtime(full_path), full_path))
                    except Exception:
                        pass
        if not candidates:
            return None, "not_found"
        candidates.sort(reverse=True)
        newest_mtime, newest_path = candidates[0]
        try:
            modal_mtime = os.path.getmtime(modal_rst)
        except Exception:
            modal_mtime = 0.0
        if newest_mtime + 1.0 < modal_mtime:
            return None, "only_stale_mcf_found:{0}".format(newest_path)
        return newest_path, "newest_under_project_root"

    def find_current_modal_rst(project_root, reported_path):
        if os.path.isfile(reported_path):
            return reported_path, "analysis_result_file_name"
        candidates = []
        for current_root, _dirs, files in os.walk(project_root):
            normalized_root = current_root.replace("\\", "/").lower()
            if "/sys-1/" in normalized_root or "/transient_" in normalized_root:
                continue
            for file_name in files:
                if file_name.lower() == "file.rst":
                    full_path = os.path.join(current_root, file_name)
                    try:
                        candidates.append((os.path.getmtime(full_path), full_path))
                    except Exception:
                        pass
        if not candidates:
            return None, "not_found:{0}".format(reported_path)
        candidates.sort(reverse=True)
        return candidates[0][1], "newest_modal_rst_under_project_root"

    summary = {"ok": False, "callback": "OnAfterSolve"}
    try:
        from Ansys.Mechanical.DataModel.Enums import DataModelObjectCategory

        transient_working_dir = os.path.abspath(analysis.WorkingDir)
        project_root = solver_root_from_working_dir(transient_working_dir)
        out_dir = transient_working_dir
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        modal_analysis = find_analysis("Modal")
        reported_modal_rst = os.path.abspath(modal_analysis.ResultFileName)
        modal_rst, modal_rst_source = find_current_modal_rst(project_root, reported_modal_rst)
        if not modal_rst:
            raise ValueError("Modal RST not found: {0}".format(modal_rst_source))
        expected_mcf_path = os.path.join(transient_working_dir, "file.mcf")
        mcf_path, mcf_source = find_current_mcf(project_root, expected_mcf_path, modal_rst)
        if not mcf_path:
            summary.update(
                {
                    "ok": False,
                    "pending": True,
                    "reason": "No current transient file.mcf is available yet.",
                    "mcf_source": mcf_source,
                    "expected_mcf": expected_mcf_path,
                    "modal_rst": modal_rst,
                    "modal_rst_source": modal_rst_source,
                    "reported_modal_rst": reported_modal_rst,
                }
            )
            write_summary(os.path.join(out_dir, OUTPUT_BASENAME + "_summary.json"), summary)
            return

        local_csys = find_coordinate_system(LOCAL_CSYS_NAME)
        requested_reference_xyz = local_csys["origin"]
        coefficient_info = force_moment_coefficients_with_dpf(
            modal_rst, requested_reference_xyz, local_csys
        )
        reference_xyz = coefficient_info.get("moment_reference_xyz", requested_reference_xyz)
        coefficients = coefficient_info["coefficients"]
        coefficient_summary = {
            "mode_count": coefficient_info.get("mode_count"),
            "element_named_selection": coefficient_info.get("element_named_selection"),
            "surface_node_named_selection": coefficient_info.get("surface_node_named_selection"),
            "surface_node_scope_method": coefficient_info.get("surface_node_scope_method"),
            "element_count": coefficient_info.get("element_count"),
            "raw_element_count": coefficient_info.get("raw_element_count"),
            "surface_node_count": coefficient_info.get("surface_node_count"),
            "side_filter": coefficient_info.get("side_filter"),
            "requested_reference_xyz": coefficient_info.get("requested_reference_xyz"),
            "moment_reference_mode": coefficient_info.get("moment_reference_mode"),
            "moment_reference_xyz": coefficient_info.get("moment_reference_xyz"),
            "modal_available_named_selections": coefficient_info.get("modal_available_named_selections"),
            "coefficients_preview": coefficients[:3],
        }
        times, modal_coordinates = parse_mcf(mcf_path)
        n_modes = min(len(coefficients), len(modal_coordinates[0]) if modal_coordinates else 0)
        if n_modes <= 0:
            raise ValueError("No common modal coordinates/coefficient modes were available.")

        resultants_global = []
        resultants_local = []
        for row in modal_coordinates:
            global_row = []
            for component_index in range(6):
                value = 0.0
                for mode_index in range(n_modes):
                    value += row[mode_index] * coefficients[mode_index][component_index]
                global_row.append(value)
            resultants_global.append(global_row)
            resultants_local.append(rotate_to_local(global_row, local_csys["axes"]))

        csv_path = os.path.join(out_dir, OUTPUT_BASENAME + ".csv")
        json_path = os.path.join(out_dir, OUTPUT_BASENAME + "_summary.json")
        summary.update(
            {
                "ok": True,
                "method": "Mechanical tree embedded DPF force_summation with construction-surface cut scoping + MCF modal-coordinate multiplication",

                "modal_rst_source": modal_rst_source,
                "reported_modal_rst": reported_modal_rst,
                "mcf": mcf_path,
                "mcf_source": mcf_source,
                "out_csv": csv_path,
                "time_point_count": len(times),
                "modes_used": n_modes,
                "modes_in_modal_rst": len(coefficients),
                "modes_in_mcf": len(modal_coordinates[0]) if modal_coordinates else 0,
                "requested_reference_xyz": requested_reference_xyz,
                "reference_xyz": reference_xyz,
                "local_coordinate_system": local_csys,
                "dpf_force_summation": coefficient_summary,
                "max_abs_resultant_global": max_abs_by_component(resultants_global),
                "max_abs_resultant_local": max_abs_by_component(resultants_local),
                "transient_rst_deleted_after_export": None,
                "transient_rst_mb_before_delete": None,
            }
        )
        try:
            write_summary(json_path, summary)
        except Exception as pre_summary_exc:
            try:
                with open(json_path + ".error.txt", "w") as stream:
                    stream.write(repr(pre_summary_exc))
                    stream.write("\n")
                    stream.write(traceback.format_exc())
                    stream.write("\n")
            except Exception:
                pass
        with open(csv_path, "wb") as stream:
            writer = csv.writer(stream)
            writer.writerow(
                [
                    "time_s",
                    "fx_global",
                    "fy_global",
                    "fz_global",
                    "mx_global",
                    "my_global",
                    "mz_global",
                    "fx_local",
                    "fy_local",
                    "fz_local",
                    "mx_local",
                    "my_local",
                    "mz_local",
                ]
            )
            for row_index, time_value in enumerate(times):
                writer.writerow(
                    [("{0:.16g}".format(time_value))]
                    + [("{0:.16g}".format(value)) for value in resultants_global[row_index]]
                    + [("{0:.16g}".format(value)) for value in resultants_local[row_index]]
                )

        transient_rst = os.path.join(transient_working_dir, "file.rst")
        deleted_transient_rst = False
        transient_rst_mb_before_delete = None
        if os.path.isfile(transient_rst):
            transient_rst_mb_before_delete = os.path.getsize(transient_rst) / 1000000.0
            try:
                os.remove(transient_rst)
                deleted_transient_rst = True
            except Exception:
                deleted_transient_rst = False

        summary.update(
            {
                "ok": True,
                "method": "Mechanical tree embedded DPF force_summation + MCF modal-coordinate multiplication",
                "modal_rst": modal_rst,
                "modal_rst_source": modal_rst_source,
                "reported_modal_rst": reported_modal_rst,
                "mcf": mcf_path,
                "mcf_source": mcf_source,
                "out_csv": csv_path,
                "out_csv_mb": os.path.getsize(csv_path) / 1000000.0,
                "time_point_count": len(times),
                "modes_used": n_modes,
                "modes_in_modal_rst": len(coefficients),
                "modes_in_mcf": len(modal_coordinates[0]) if modal_coordinates else 0,
                "requested_reference_xyz": requested_reference_xyz,
                "reference_xyz": reference_xyz,
                "local_coordinate_system": local_csys,
                "dpf_force_summation": coefficient_summary,
                "max_abs_resultant_global": max_abs_by_component(resultants_global),
                "max_abs_resultant_local": max_abs_by_component(resultants_local),
                "transient_rst_deleted_after_export": deleted_transient_rst,
                "transient_rst_mb_before_delete": transient_rst_mb_before_delete,
            }
        )
        write_summary(json_path, summary)
    except Exception as exc:
        summary.update(error=repr(exc), traceback=traceback.format_exc())
