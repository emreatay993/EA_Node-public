# MCF section resultant controls for the Modal analysis.
# Target Callback must be "Get Solve Commands".
# This keeps modal ENF coefficients for the selected side elements without writing
# a transient time-history RST later.

apdl_lines = [
    "/COM, MCF_SECTION_MODAL_BEGIN",
    "/nopr",
    "cmsel,s,DENEME,elem",
    "*get,_section_elem_count,elem,0,count",
    "*if,_section_elem_count,le,0,then",
    "  /gopr",
    "  /com,ERROR: DENEME selected zero elements; modal section coefficients were not configured.",
    "*else",
    "  cm,_section_elems,elem",
    "  nsle,s,1",
    "  *get,_section_node_count,node,0,count",
    "  cm,_section_nodes,node",
    "  allsel,all",
    "  /gopr",
    "  outres,erase",
    "  outres,all,none",
    "  outres,nsol,all,_section_nodes",
    "  outres,nload,all,_section_elems",
    "  mxpand,,,,yes,,no",
    "  /com,MCF modal section coefficient output active: elements=_SECTION_ELEMS nodes=_SECTION_NODES.",
    "*endif",
    "allsel,all",
    "/gopr",
    "/COM, MCF_SECTION_MODAL_END",
]

for apdl_line in apdl_lines:
    solver_input_file.WriteLine(apdl_line)
